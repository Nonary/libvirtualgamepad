cmake_minimum_required(VERSION 3.20)
project(VibeshineMouseMapperCore LANGUAGES CXX)
enable_testing()
add_executable(test_mouse_mapper test_mapper.cpp)
target_include_directories(test_mouse_mapper PRIVATE ../../include)
target_compile_features(test_mouse_mapper PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(test_mouse_mapper PRIVATE /W4 /WX)
else()
  target_compile_options(test_mouse_mapper PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME mouse_mapper_conversion_and_backpressure COMMAND test_mouse_mapper)
find_package(Threads REQUIRED)
add_executable(test_mouse_mapper_native test_mapper_native.cpp)
target_link_libraries(test_mouse_mapper_native PRIVATE Threads::Threads)
target_include_directories(test_mouse_mapper_native PRIVATE ../../include)
target_compile_features(test_mouse_mapper_native PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(test_mouse_mapper_native PRIVATE /W4 /WX)
else()
  target_compile_options(test_mouse_mapper_native PRIVATE -Wall -Wextra -Werror -pedantic -Wno-multichar)
endif()
add_test(NAME mouse_mapper_native_callback_contract COMMAND test_mouse_mapper_native)

set_tests_properties(mouse_mapper_native_callback_contract PROPERTIES TIMEOUT 20)
